(() => {
  const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
  const work = async () => {
    const [{ PRESETS, buildPreset }, { generateProject }] = await Promise.all([
      import('/src/lib/presets.ts'),
      import('/src/lib/pipeline.ts'),
    ])
    const out = {}
    const p = PRESETS.find((x) => x.id === 'gold-dust')
    const a = buildPreset(p)
    const b = buildPreset(p)
    out.seedA = a.seed
    out.seedB = b.seed
    out.sameSeed = a.seed === b.seed
    out.projectJsonEqual = JSON.stringify(a) === JSON.stringify(b)

    const r1 = await generateProject(a, { useCache: false })
    const r2 = await generateProject(a, { useCache: false })
    const h = (rs) => rs.map((r) => r.ir.nodes.length).join(',')
    out.sameProjectTwoRenders = h(r1) === h(r2)
    out.counts1 = h(r1)
    out.counts2 = h(r2)
    const s1 = JSON.stringify(r1.map((r) => r.ir.nodes))
    const s2 = JSON.stringify(r2.map((r) => r.ir.nodes))
    out.contentEqual = s1 === s2
    out.len1 = s1.length
    out.len2 = s2.length

    const r3 = await generateProject(b, { useCache: false })
    out.buildTwiceContentEqual = s1 === JSON.stringify(r3.map((x) => x.ir.nodes))

    // is Math.random in play?
    const m1 = JSON.stringify(Array.from({ length: 5 }, () => Math.random()))
    const m2 = JSON.stringify(Array.from({ length: 5 }, () => Math.random()))
    out.mathRandomDiffers = m1 !== m2
    return JSON.stringify(out, null, 1)
  }
  return Promise.race([work(), sleep(60000).then(() => 'TIMEOUT-60s')])
})()
