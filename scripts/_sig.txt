(() => {
  const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
  const work = async () => {
    const [{ buildIR }, { PRESETS, buildPreset }, { generateProject }, { drawIR }] =
      await Promise.all([
        import('/src/lib/ir.ts'),
        import('/src/lib/presets.ts'),
        import('/src/lib/pipeline.ts'),
        import('/src/lib/render/canvas.ts'),
      ])
    const W = 1600
    const H = 1000
    const IDS = [
      'prism-spirograph',
      'rain-on-lens',
      'spiral-galaxy',
      'aurora-veil',
      'vapor-bloom',
      'gold-dust',
      'sunset-bokeh-field',
      'diamond-dust',
    ]
    const out = {}
    for (const id of IDS) {
      const p = PRESETS.find((x) => x.id === id)
      const project = buildPreset(p)
      const results = await generateProject(project, { useCache: false })
      const nodes = results.flatMap((r) => r.ir.nodes)
      const ir = buildIR(project.canvas.w, project.canvas.h, nodes)
      const c = document.createElement('canvas')
      c.width = W
      c.height = H
      const x = c.getContext('2d', { willReadFrequently: true })
      drawIR(x, ir, 0.5)
      const d = x.getImageData(0, 0, W, H).data
      // 32x32 block means — stable under tiny resampling differences, but a
      // real spatial/colour shift shows up immediately
      const BX = 32
      const BY = 32
      const bw = Math.floor(W / BX)
      const bh = Math.floor(H / BY)
      const sig = new Array(BX * BY * 4).fill(0)
      for (let by = 0; by < BY; by++) {
        for (let bx = 0; bx < BX; bx++) {
          let r = 0
          let g = 0
          let b = 0
          let a = 0
          let n = 0
          for (let py = by * bh; py < (by + 1) * bh; py++) {
            let i = (py * W + bx * bw) * 4
            for (let px = 0; px < bw; px++, i += 4) {
              r += d[i]
              g += d[i + 1]
              b += d[i + 2]
              a += d[i + 3]
              n++
            }
          }
          const o = (by * BX + bx) * 4
          sig[o] = +(r / n).toFixed(3)
          sig[o + 1] = +(g / n).toFixed(3)
          sig[o + 2] = +(b / n).toFixed(3)
          sig[o + 3] = +(a / n).toFixed(3)
        }
      }
      out[id] = sig
    }
    return JSON.stringify(out)
  }
  return Promise.race([work(), sleep(120000).then(() => 'TIMEOUT-120s')])
})()
